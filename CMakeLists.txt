cmake_minimum_required(VERSION 3.20)
project(InkBallReconstructed VERSION 1.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(inkball_engine STATIC src/Game.cpp src/Levels.cpp src/BoardCatalog.cpp)
target_include_directories(inkball_engine PUBLIC src)
if(MSVC)
  set_property(TARGET inkball_engine PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(inkball_engine PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(inkball_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
  enable_language(RC)
  # MSVC/CMake's manifest tool owns embedding; the shared .rc has no manifest.
  # MinGW has no mt.exe step, so only that toolchain gets a generated resource.
  if(MSVC)
    set(INKBALL_MANIFEST_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.manifest")
  elseif(MINGW)
    set(INKBALL_MANIFEST_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/inkball-manifest.rc")
    file(GENERATE OUTPUT "${INKBALL_MANIFEST_SOURCE}"
      CONTENT "1 24 \"${CMAKE_CURRENT_SOURCE_DIR}/assets/app.manifest\"\n")
  endif()
  add_executable(inkball WIN32 src/WinMain.cpp src/Renderer.cpp src/BoardBrowser.cpp
    src/MechanicsDialog.cpp src/PenCursor.cpp
    resources/InkBall.rc ${INKBALL_MANIFEST_SOURCE})
  target_compile_definitions(inkball PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_include_directories(inkball PRIVATE src resources)
  target_link_libraries(inkball PRIVATE inkball_engine user32 gdi32 msimg32 shell32 comctl32)
  if(MSVC)
    set_property(TARGET inkball PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(inkball PRIVATE /W4 /permissive- /utf-8)
  elseif(MINGW)
    target_link_options(inkball PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(inkball_tests tests/engine_tests.cpp)
  target_link_libraries(inkball_tests PRIVATE inkball_engine)
  if(MSVC)
    set_property(TARGET inkball_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(inkball_tests PRIVATE /W4 /permissive- /utf-8)
  elseif(MINGW)
    target_link_options(inkball_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME engine COMMAND inkball_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets/levels.ikl")

  add_executable(inkball_catalog_tests tests/catalog_tests.cpp)
  target_link_libraries(inkball_catalog_tests PRIVATE inkball_engine)
  if(MSVC)
    set_property(TARGET inkball_catalog_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(inkball_catalog_tests PRIVATE /W4 /permissive- /utf-8)
  elseif(MINGW)
    target_link_options(inkball_catalog_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME catalog COMMAND inkball_catalog_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets/levels.ikl")

  if(WIN32)
    add_executable(inkball_renderer_tests tests/renderer_tests.cpp src/Renderer.cpp
      resources/InkBall.rc ${INKBALL_MANIFEST_SOURCE})
    target_compile_definitions(inkball_renderer_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_include_directories(inkball_renderer_tests PRIVATE src resources)
    target_link_libraries(inkball_renderer_tests PRIVATE inkball_engine user32 gdi32 msimg32)
    if(MSVC)
      set_property(TARGET inkball_renderer_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
      target_compile_options(inkball_renderer_tests PRIVATE /W4 /permissive- /utf-8)
    elseif(MINGW)
      target_link_options(inkball_renderer_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME renderer COMMAND inkball_renderer_tests)
  endif()
endif()
